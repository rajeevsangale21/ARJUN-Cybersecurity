============================================================ # HELPER FUNCTIONS # ============================================================ def load_input_file(uploaded_file): suffix = Path( uploaded_file.name ).suffix.lower() temp_dir = ( PROJECT_ROOT / "data" / "sample" ) temp_dir.mkdir( parents=True, exist_ok=True ) temp_path = ( temp_dir / uploaded_file.name ) with open( temp_path, "wb" ) as file: file.write( uploaded_file.getbuffer() ) return temp_path def process_input(path): suffix = path.suffix.lower() if suffix == ".csv": from ingestion.csv_loader import ( load_csv ) return load_csv(path) if suffix in [".pcap", ".pcapng"]: from ingestion.pcap_loader import ( load_pcap ) return load_pcap(path) if suffix == ".log": from ingestion.log_loader import ( load_generic_log ) return load_generic_log(path) if suffix == ".txt": from ingestion.log_loader import ( load_generic_log ) return load_generic_log(path) raise ValueError( f"Unsupported file type: {suffix}" ) def run_arjun( input_path, steps, sequence_length ): # -------------------------------------------------------- # Import pipeline here so the dashboard loads faster # -------------------------------------------------------- from pipeline import run_pipeline result = run_pipeline( str(input_path), sequence_length=sequence_length ) states = result["states"] window_ids = result["window_ids"] df = result["data"] if len(states) < sequence_length: raise ValueError( "Not enough network states for " "the selected sequence length." ) # -------------------------------------------------------- # Normalize # -------------------------------------------------------- normalizer = ( NetworkStateNormalizer() ) normalizer.load( NORMALIZER_PATH ) normalized_states = ( normalizer.transform( states ) ) # -------------------------------------------------------- # Graphs # -------------------------------------------------------- graphs = build_window_graphs( df ) # -------------------------------------------------------- # Device # -------------------------------------------------------- device = ( "cuda" if torch.cuda.is_available() else "cpu" ) # -------------------------------------------------------- # XGBoost # -------------------------------------------------------- classifier = ( XGBoostAttackClassifier.load( XGBOOST_PATH ) ) current_state = ( normalized_states[-1] ) current_probability = float( classifier.predict_proba( current_state.reshape( 1, -1 ) )[0, 1] ) # -------------------------------------------------------- # World Model # -------------------------------------------------------- world_model = ( HybridWorldModelWrapper.load( WORLD_MODEL_PATH, device=device ) ) state_sequence = ( normalized_states[ -sequence_length: ] ) graph_ids = window_ids[ -sequence_length: ] graph_sequence = [] for window_id in graph_ids: window_id = int( window_id ) if window_id not in graphs: raise ValueError( f"Graph not found for " f"window {window_id}" ) graph = graphs[ window_id ] graph_sequence.append( { "node_features": torch.tensor( graph["node_features"], dtype=torch.float32, device=device ), "adjacency": torch.tensor( graph["adjacency"], dtype=torch.float32, device=device ) } ) # -------------------------------------------------------- # Forecast # -------------------------------------------------------- forecaster = ( HybridKStepForecaster( model=world_model, device=device ) ) future_states = ( forecaster.forecast( state_sequence, graph_sequence, steps=steps ) ) # -------------------------------------------------------- # Unified analysis # -------------------------------------------------------- encoder = ( NetworkStateEncoder() ) analysis_service = ( UnifiedAnalysisService( encoder.feature_names ) ) analysis = ( analysis_service.analyze( current_state=current_state, future_states=future_states, current_attack_probability=( current_probability ) ) ) return { "states": states, "normalized_states": normalized_states, "window_ids": window_ids, "data": df, "future_states": future_states, "current_probability": current_probability, "analysis": analysis, "feature_names": encoder.feature_names } def risk_number(level): levels = { "LOW": 1, "MEDIUM": 2, "HIGH": 3, "CRITICAL": 4 } return levels.get( level, 0 ) # ============================================================ # RUN ANALYSIS # ============================================================ if uploaded_file: st.sidebar.success( f"Loaded: {uploaded_file.name}" ) run_button = st.sidebar.button( "🚀 Analyze Network", use_container_width=True ) if run_button: input_path = None try: with st.spinner( "ARJUN is analyzing the network..." ): input_path = ( load_input_file( uploaded_file ) ) result = run_arjun( input_path, forecast_steps, sequence_length ) st.session_state.analysis_result = ( result ) st.session_state.raw_dataframe = ( result["data"] ) st.success( "Analysis completed successfully." ) except Exception as error: st.error( f"Analysis failed: {error}" ) st.exception(error) #
